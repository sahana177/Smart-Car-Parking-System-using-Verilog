module smart_parking (
    input clk,
    input reset,
    input entry,
    input exit,
    output reg [3:0] car_count,
    output LED
);

    parameter MAX_CAPACITY = 4;

    always @(posedge clk or posedge reset) begin
        if (reset)
            car_count <= 4'd0;

        else begin
            case ({entry, exit})

                2'b10: begin
                    if (car_count < MAX_CAPACITY)
                        car_count <= car_count + 1'b1;
                end

                2'b01: begin
                    if (car_count > 0)
                        car_count <= car_count - 1'b1;
                end

                2'b11: begin
                    car_count <= car_count;
                end

                2'b00: begin
                    car_count <= car_count;
                end

            endcase
        end
    end
    assign LED = (car_count == MAX_CAPACITY);

endmodule
